import pandas as pd
import numpy as np

# np.random.seed(42)
n=30
df= pd.DataFrame({
    'Math_Score': np.random.randint(60,81,size = n),
    'Reading_Score':np.random.randint(60,81,size=n),
    'Writing_Score':   np.random.randint(60, 81, size=n),
    'Placement_Score': np.random.randint(60, 81, size=n),
    'Club_Join_Date':  np.random.randint(2018, 2022, size=n)
})

imp_idx = np.random.choice(n,size = int(0.2*n),replace = False)
df.loc[imp_idx[:5],'Math_Score'] = np.random.choice([50,90],size = 5).astype('int32')
df.loc[imp_idx[5:6],'Reading_Score'] = np.nan
df.loc[imp_idx[:5],'Writing_Score'] = np.random.choice([55,85],size = 5).astype('int32')
df.loc[imp_idx[:5],'Club_Join_Date'] = np.random.choice([2015,2025],size = 5).astype('int32')

df.isnull().sum()

df['Reading_Score'] = df['Reading_Score'].interpolate()
df

numeric_cols = ['Math_Score','Reading_Score','Writing_Score','Placement_Score','Club_Join_Date']

for col in numeric_cols:
    Q1,Q3 = df[col].quantile([0.25,0.75])
    IQR = Q3 - Q1
    lower = Q1 - 1.5*IQR
    upper = Q3 + 1.5*IQR
    mask = (df[col]<lower) | (df[col]>upper)
    if mask.any():
        df.loc[mask,col]  = df[col].median().astype('int32')

df.describe()
